@st.cache_data(show_spinner=False)
def run_cached_simulation(
    disruption_active: bool,
    transit_delay: float,
    demand_shock: float,
    ambient_temp_c: float,
    min_buffer: float,
    max_discount: float,
    solver_mode: Literal["urgency_margin", "lp_highs"],
) -> Tuple[OptimizationSummary, OptimizationSummary, UncertaintyReport, pd.DataFrame]:
    """Cached simulation pipeline for O(1) UI responsiveness across widget interactions."""
    batches, destinations, routes = get_standard_scenario()
    if ambient_temp_c > 4.0:
        batches = apply_temperature_excursion(batches, ambient_temp_c=ambient_temp_c)
    if disruption_active:
        routes, destinations = apply_disruption(
            routes,
            destinations,
            delay_hours=transit_delay,
            demand_shock_pct=-demand_shock,
        )

    baseline_solver = BaselineSolver()
    fresh_solver = FreshRouteSolver(
        min_safe_buffer_hrs=min_buffer,
        max_discount_pct=max_discount,
        solver_mode=solver_mode,
    )

    base_res = baseline_solver.solve(batches, destinations, routes)
    fresh_res = fresh_solver.solve(batches, destinations, routes)
    mc_report = fresh_solver.evaluate_uncertainty(batches, destinations, routes, runs=40)
    batch_df = pd.DataFrame([b.model_dump() for b in batches])
    return base_res, fresh_res, mc_report, batch_df

# FreshRoute: Intelligent Decision Support for Perishable Produce Distribution

## 1. Executive Summary & Problem Context

India's Ministry of Food Processing Industries (MoFPI) reported severe post-harvest losses across 54 commodities in 15 agro-climatic zones, with fruits and vegetables suffering 15% to 25% waste between farm gates and retail markets. 

Traditional agricultural distributors rely on naive FIFO (First-In, First-Out) dispatching the oldest inventory to the first available destination order. This causes three critical supply chain failures:
1. In-Transit & Dock Spoilage: Batches with short shelf life are sent on long transit corridors and arrive with insufficient commercial shelf life (< 6.0h), causing retailers to reject and dump consignments at the loading dock.
2. Suboptimal Revenue Realization: High-perishability produce is dumped into low-price local markets instead of high-margin transit corridors.
3. Binary Pricing (Spoil or Sell): Produce is priced at 100% until it spoils, instead of applying dynamic clearance discounting to salvage near-expiry batches.

FreshRoute is a constrained decision-support engine that unifies remaining shelf-life estimates, transport matrices, dynamic pricing, and destination demand to produce a mathematically optimal, 100% explainable, and zero-in-transit-spoilage dispatch schedule.

## 3. Core Technical Architecture

FreshRoute is engineered across three decoupled layers:

```
[ Raw Inputs: Batches, Mandi Demands, Routes ]
                       |
                       v
+-------------------------------------------------------------+
| 1. Constraint & Feasibility Layer (models.py)               |
|    - Hard Bounds: S_i >= T_ij + H_j + Buffer (2.0h)         |
|    - Transport Vehicle Capacity & Mandi Demand Caps         |
+-------------------------------------------------------------+
                       |
                       v
+-------------------------------------------------------------+
| 2. Intelligent Optimization Engine (freshroute_solver.py)   |
|    - Multi-Objective Priority Index (Margin x Urgency)      |
|    - Dynamic Salvage Discounting (0% - 25% Decay Curve)     |
+-------------------------------------------------------------+
                       |
                       v
+-------------------------------------------------------------+
| 3. Audit & Explainability Layer (explainability.py)         |
|    - Line-item trace: Base cost, Freight, Net Margin/kg     |
|    - Evidence-backed reasoning for every single kilogram    |
+-------------------------------------------------------------+
                       |
         +-------------+-------------+
         v                           v
[ CLI Benchmark (run_demo.py) ]  [ Interactive Dashboard (app.py) ]
```

### Key Engineering Safeguards:
1. Zero In-Transit Spoilage Guarantee: Enforces strict inequality S_i >= T_ij + H_j + safe_buffer, where S_i is batch shelf life, T_ij is route transit time, H_j is destination handling time, and safe_buffer is 2.0h minimum arrival buffer.
2. Targeted Dynamic Clearance Discounting: When arrival buffer is between 2h and 12h, the engine dynamically applies a calibrated clearance discount (up to 25%) to ensure immediate mandi turnover and prevent commercial rejection.
3. Traceable Decision Audit: Every dispatch includes a transparent mathematical proof showing gross revenue, freight cost per kg, applied discount, and exact business justification.

"""engine/baseline_solver.py: Realistic Naive FIFO Dispatcher with O(B + D) pre-indexed lookup."""

from collections import defaultdict
import time
from typing import Dict, List

from .models import (
    AllocationDecision,
    Batch,
    Destination,
    OptimizationSummary,
    TransportRoute,
)


class BaselineSolver:
    """
    Standard naive baseline:
    - Dispatches oldest batches first (FIFO) to fulfill orders in order of destination appearance.
    - Zero dynamic discounting.
    - Retail dock inspection: Shipments arriving with < 6.0h shelf-life buffer without clearance
      discounting fail commercial inspection and are rejected at the dock as spoiled.
    """

    def __init__(self, min_retail_buffer_hrs: float = 6.0, spoilage_penalty_per_kg: float = 5.0):
        if min_retail_buffer_hrs < 0:
            raise ValueError("min_retail_buffer_hrs must be non-negative.")
        if spoilage_penalty_per_kg < 0:
            raise ValueError("spoilage_penalty_per_kg must be non-negative.")
        self.min_retail_buffer_hrs = min_retail_buffer_hrs
        self.spoilage_penalty_per_kg = spoilage_penalty_per_kg

    def solve(
        self,
        batches: List[Batch],
        destinations: List[Destination],
        routes: List[TransportRoute],
    ) -> OptimizationSummary:
        start_ns = time.perf_counter_ns()

        route_map = {(r.origin, r.dest_id): r for r in routes}
        remaining_demand = {d.dest_id: d.demand_kg for d in destinations}
        total_inventory_kg = sum(b.quantity_kg for b in batches)

        # Pre-index destinations by commodity for O(1) lookup instead of O(B * D) filtering
        dests_by_commodity: Dict[str, List[Destination]] = defaultdict(list)
        for d in destinations:
            dests_by_commodity[d.commodity].append(d)

        # FIFO: oldest batches dispatched first
        sorted_batches = sorted(batches, key=lambda b: b.remaining_shelf_life_hrs)
        allocations: List[AllocationDecision] = []
        dispatched_kg = 0.0
        spoiled_kg = 0.0

        for batch in sorted_batches:
            batch_qty = batch.quantity_kg
            candidate_dests = dests_by_commodity.get(batch.commodity, [])

            for dest in candidate_dests:
                if batch_qty <= 0:
                    break

                avail_demand = remaining_demand[dest.dest_id]
                if avail_demand <= 0:
                    continue

                route = route_map.get((batch.origin, dest.dest_id))
                if not route:
                    continue

                alloc_qty = min(batch_qty, avail_demand)
                if alloc_qty <= 0:
                    continue

                travel_time = route.travel_time_hrs
                total_time_to_shelf = travel_time + dest.handling_time_hrs
                buffer = batch.remaining_shelf_life_hrs - total_time_to_shelf

                discount = 0.0
                final_price = dest.selling_price_per_kg

                if buffer < self.min_retail_buffer_hrs:
                    spoiled_kg += alloc_qty
                    sold_qty = 0.0
                    clearance_rate = 0.0
                    reason = (
                        f"REJECTED/SPOILED: Arrived with insufficient buffer "
                        f"({buffer:.1f}h < {self.min_retail_buffer_hrs}h contracted window)."
                    )
                    gross_rev = 0.0
                else:
                    sold_qty = alloc_qty
                    clearance_rate = 1.0
                    gross_rev = alloc_qty * final_price
                    reason = f"Dispatched via naive FIFO to {dest.name}."

                t_cost = alloc_qty * route.cost_per_kg
                p_cost = alloc_qty * batch.base_cost_per_kg
                net = gross_rev - t_cost - p_cost

                allocations.append(
                    AllocationDecision(
                        batch_id=batch.batch_id,
                        dest_id=dest.dest_id,
                        commodity=batch.commodity,
                        quantity_kg=alloc_qty,
                        sold_qty_kg=sold_qty,
                        clearance_rate=clearance_rate,
                        travel_time_hrs=travel_time,
                        remaining_shelf_life_hrs=batch.remaining_shelf_life_hrs,
                        shelf_life_buffer_hrs=round(buffer, 1),
                        base_price_per_kg=dest.selling_price_per_kg,
                        discount_pct=discount,
                        final_price_per_kg=final_price,
                        transport_cost_per_kg=route.cost_per_kg,
                        base_cost_per_kg=batch.base_cost_per_kg,
                        gross_revenue=round(gross_rev, 2),
                        transport_cost_total=round(t_cost, 2),
                        procurement_cost_total=round(p_cost, 2),
                        net_profit=round(net, 2),
                        reason=reason,
                    )
                )

                remaining_demand[dest.dest_id] -= alloc_qty
                batch_qty -= alloc_qty
                dispatched_kg += alloc_qty

            if batch_qty > 0 and batch.remaining_shelf_life_hrs <= 24.0:
                spoiled_kg += batch_qty

        unsold_unspoiled = max(0.0, total_inventory_kg - dispatched_kg)
        tot_gross = sum(a.gross_revenue for a in allocations)
        tot_trans = sum(a.transport_cost_total for a in allocations)
        tot_proc = sum(a.procurement_cost_total for a in allocations)
        waste_penalty = spoiled_kg * self.spoilage_penalty_per_kg
        net_profit = tot_gross - tot_trans - tot_proc - waste_penalty

        total_demand = sum(d.demand_kg for d in destinations)
        fulfilled = sum(a.sold_qty_kg for a in allocations)
        fulfillment_pct = (fulfilled / total_demand * 100.0) if total_demand > 0 else 0.0
        spoilage_pct = (spoiled_kg / total_inventory_kg * 100.0) if total_inventory_kg > 0 else 0.0
        elapsed_ms = (time.perf_counter_ns() - start_ns) / 1_000_000.0

        return OptimizationSummary(
            strategy_name="Baseline (Naive FIFO)",
            total_inventory_kg=total_inventory_kg,
            dispatched_kg=dispatched_kg,
            spoiled_kg=round(spoiled_kg, 1),
            unsold_unspoiled_kg=round(unsold_unspoiled, 1),
            spoilage_rate_pct=round(spoilage_pct, 2),
            demand_fulfillment_pct=round(fulfillment_pct, 2),
            total_gross_revenue=round(tot_gross, 2),
            total_transport_cost=round(tot_trans, 2),
            total_procurement_cost=round(tot_proc, 2),
            disposal_waste_cost=round(waste_penalty, 2),
            net_profit=round(net_profit, 2),
            solve_time_ms=round(elapsed_ms, 3),
            allocations=allocations,
        )

def apply_disruption(
    routes: List[TransportRoute],
    destinations: List[Destination],
    delay_hours: float = 6.0,
    demand_shock_pct: float = -0.30,
    capacity_reduction_pct: float = 0.0,
) -> Tuple[List[TransportRoute], List[Destination]]:
    """Simulates realistic supply-chain complications (highway delays, demand shocks, fleet reduction)."""
    if delay_hours < 0:
        raise ValueError("delay_hours must be non-negative.")
    if not -1.0 <= demand_shock_pct <= 5.0:
        raise ValueError("demand_shock_pct must be between -1.0 and 5.0.")
    if not 0.0 <= capacity_reduction_pct < 1.0:
        raise ValueError("capacity_reduction_pct must be in [0.0, 1.0).")

    disrupted_routes = [
        r.model_copy(
            update={
                "travel_time_hrs": round(r.travel_time_hrs + delay_hours, 2),
                "vehicle_capacity_kg": max(1.0, round(r.vehicle_capacity_kg * (1.0 - capacity_reduction_pct), 2)),
            }
        )
        for r in routes
    ]

    disrupted_dests = [
        d.model_copy(
            update={
                "demand_kg": max(0.0, round(d.demand_kg * (1.0 + demand_shock_pct), 2)),
            }
        )
        for d in destinations
    ]

    return disrupted_routes, disrupted_dests


def apply_temperature_excursion(
    batches: List[Batch],
    ambient_temp_c: float = 4.0,
    reference_temp_c: float = 4.0,
    q10_factor: float = 2.0,
) -> List[Batch]:
    """
    Adjusts batch remaining shelf life using the biological Q10 thermal respiration law:
    Effective Shelf Life = Nominal Shelf Life / (Q10 ^ ((T_actual - T_ref) / 10)).
    """
    if ambient_temp_c <= reference_temp_c:
        return [b.model_copy(update={"storage_temp_c": ambient_temp_c}) for b in batches]

    acceleration = q10_factor ** ((ambient_temp_c - reference_temp_c) / 10.0)
    adjusted_batches: List[Batch] = []
    for b in batches:
        # Storable dry bulbs (Onion) are less sensitive to cold-chain excursion than wet produce
        effective_accel = 1.0 + (acceleration - 1.0) * (0.25 if b.commodity == "Onion" else 1.0)
        new_shelf_life = max(1.0, round(b.remaining_shelf_life_hrs / effective_accel, 1))
        adjusted_batches.append(
            b.model_copy(
                update={
                    "remaining_shelf_life_hrs": new_shelf_life,
                    "storage_temp_c": ambient_temp_c,
                }
            )
        )
    return adjusted_batches

"""engine/explainability.py: Evidence-backed decision explanation, formatting, and CSV sanitization."""

from typing import Any, Dict, List
import pandas as pd

def format_inr(number: float, prefix: str = "\u20b9", decimals: int = 0) -> str:
    """Formats a float into the Indian Numbering System (Lakhs, Crores) without code duplication."""
    neg = "-" if number < 0 else ""
    num_abs = abs(float(number))
    formatted_float = f"{num_abs:.{decimals}f}"
    parts = formatted_float.split(".")
    integer_part = parts[0]
    decimal_part = f".{parts[1]}" if decimals > 0 else ""

    if len(integer_part) <= 3:
        res = integer_part
    else:
        last3 = integer_part[-3:]
        remaining = integer_part[:-3]
        groups: List[str] = []
        while len(remaining) > 2:
            groups.insert(0, remaining[-2:])
            remaining = remaining[:-2]
        if remaining:
            groups.insert(0, remaining)
        res = ",".join(groups) + "," + last3

    return f"{prefix}{neg}{res}{decimal_part}" if prefix else f"{neg}{res}{decimal_part}"


def sanitize_csv_value(val: Any) -> Any:
    """
    Security safeguard: Neutralizes spreadsheet formula injection (CSV Injection)
    by prefixing strings starting with '=', '+', '-', '@', tab, or CR with a single quote.
    """
    if isinstance(val, str) and val.startswith(("=", "+", "-", "@", "\t", "\r")):
        return f"'{val}"
    return val

"""engine/freshroute_solver.py: Constrained Optimization Solver with Clearance Elasticity & LP Support."""

from collections import defaultdict
import time
from typing import Any, Dict, List, Literal, Tuple

import numpy as np
from scipy.optimize import linprog

from .models import (
    AllocationDecision,
    Batch,
    Destination,
    OptimizationSummary,
    TransportRoute,
    UncertaintyReport,
)


class FreshRouteSolver:
    """
    Intelligent Decision Engine:
    1. Feasibility: Enforces remaining_shelf_life >= travel_time + handling_time + min_safe_buffer.
    2. Targeted Dynamic Clearance: Applies calibrated discounts (0-25%) when arrival buffer < 12h
       to stimulate rapid mandi absorption and prevent commercial dock rejection.
    3. Clearance Elasticity: Models buyer absorption velocity when discounts are constrained.
    4. Dual Solver Support:
       - 'urgency_margin': Priority-Index greedy dispatch (fast, deterministic explainability).
       - 'lp_highs': Exact Linear Programming via SciPy HiGHS simplex/interior-point solver.
    """

    def __init__(
        self,
        min_safe_buffer_hrs: float = 2.0,
        discount_threshold_hrs: float = 12.0,
        max_discount_pct: float = 0.25,
        spoilage_penalty_per_kg: float = 5.0,
        solver_mode: Literal["urgency_margin", "lp_highs"] = "urgency_margin",
    ):
        if min_safe_buffer_hrs < 0:
            raise ValueError("min_safe_buffer_hrs must be non-negative.")
        if discount_threshold_hrs <= 0:
            raise ValueError("discount_threshold_hrs must be positive.")
        if not 0.0 <= max_discount_pct <= 1.0:
            raise ValueError("max_discount_pct must be between 0.0 and 1.0.")
        if spoilage_penalty_per_kg < 0:
            raise ValueError("spoilage_penalty_per_kg must be non-negative.")

        self.min_safe_buffer_hrs = min_safe_buffer_hrs
        self.discount_threshold_hrs = discount_threshold_hrs
        self.max_discount_pct = max_discount_pct
        self.spoilage_penalty_per_kg = spoilage_penalty_per_kg
        self.solver_mode = solver_mode

    def calculate_discount_and_clearance(self, buffer_hrs: float) -> Tuple[float, float, str]:
        """
        Calculates dynamic discount and expected market clearance rate.
        Disclosed Rule:
        - When arrival buffer >= 12.0h, 0% discount is required (100% clearance).
        - When 2.0h <= arrival buffer < 12.0h, urgency scales linearly:
          urgency = (12.0 - buffer) / 12.0.
        - Market requires at least `urgency * 20%` discount for 100% immediate absorption.
        """
        if buffer_hrs <= 0:
            return 0.0, 0.0, "EXPIRED"
        if buffer_hrs >= self.discount_threshold_hrs:
            return 0.0, 1.0, "Full Price (Adequate Buffer)"

        urgency = (self.discount_threshold_hrs - buffer_hrs) / self.discount_threshold_hrs
        required_clearance_discount = urgency * 0.20
        actual_discount = round(urgency * self.max_discount_pct, 2)

        if actual_discount >= required_clearance_discount:
            clearance_rate = 1.0
            reason = (
                f"Dynamic clearance {int(actual_discount * 100)}% applied: "
                "100% inventory cleared in time"
            )
        else:
            shortfall = required_clearance_discount - actual_discount
            clearance_rate = max(0.50, round(1.0 - (shortfall * 2.5), 2))
            reason = (
                f"Under-discounted ({int(actual_discount * 100)}% vs "
                f"{int(required_clearance_discount * 100)}% req): "
                f"only {int(clearance_rate * 100)}% cleared before expiry"
            )

        return actual_discount, clearance_rate, reason

    def _build_candidates(
        self,
        batches: List[Batch],
        destinations: List[Destination],
        route_map: Dict[Tuple[str, str], TransportRoute],
    ) -> List[Dict[str, Any]]:
        """Builds feasible (batch, destination) candidate arcs using O(B + D) commodity indexing."""
        dests_by_commodity: Dict[str, List[Destination]] = defaultdict(list)
        for d in destinations:
            dests_by_commodity[d.commodity].append(d)

        candidates: List[Dict[str, Any]] = []
        for b in batches:
            for d in dests_by_commodity.get(b.commodity, []):
                route = route_map.get((b.origin, d.dest_id))
                if not route:
                    continue

                total_transit_hrs = route.travel_time_hrs + d.handling_time_hrs
                buffer_hrs = b.remaining_shelf_life_hrs - total_transit_hrs
                if buffer_hrs < self.min_safe_buffer_hrs:
                    continue

                discount_pct, clearance_rate, discount_reason = (
                    self.calculate_discount_and_clearance(buffer_hrs)
                )
                effective_price = d.selling_price_per_kg * (1.0 - discount_pct)
                unit_margin = (
                    (effective_price * clearance_rate)
                    - route.cost_per_kg
                    - b.base_cost_per_kg
                    - ((1.0 - clearance_rate) * self.spoilage_penalty_per_kg)
                )

                if unit_margin <= 0:
                    continue

                urgency_multiplier = 1.0 + (3.0 / max(b.remaining_shelf_life_hrs, 1.0))
                priority_index = unit_margin * urgency_multiplier

                candidates.append(
                    {
                        "batch": b,
                        "dest": d,
                        "route": route,
                        "buffer_hrs": buffer_hrs,
                        "discount_pct": discount_pct,
                        "clearance_rate": clearance_rate,
                        "discount_reason": discount_reason,
                        "effective_price": effective_price,
                        "unit_margin": effective_price - route.cost_per_kg - b.base_cost_per_kg,
                        "net_unit_contribution": unit_margin,
                        "priority_index": priority_index,
                    }
                )

        candidates.sort(key=lambda x: x["priority_index"], reverse=True)
        return candidates

    def _allocate_lp(
        self,
        candidates: List[Dict[str, Any]],
        batches: List[Batch],
        destinations: List[Destination],
        routes: List[TransportRoute],
    ) -> List[Tuple[Dict[str, Any], float]]:
        """Solves exact global allocation using SciPy HiGHS Linear Programming."""
        if not candidates:
            return []

        n_vars = len(candidates)
        # Maximize priority-weighted net contribution (avoiding perishable warehouse expiration penalties)
        c = np.array(
            [
                -(
                    cand["net_unit_contribution"]
                    + (self.spoilage_penalty_per_kg if cand["batch"].remaining_shelf_life_hrs <= 24.0 else 0.0)
                    + 1e-3 * cand["priority_index"]
                )
                for cand in candidates
            ],
            dtype=float,
        )

        rows: List[np.ndarray] = []
        rhs: List[float] = []

        # 1. Batch supply constraints
        for b in batches:
            row = np.zeros(n_vars, dtype=float)
            has_arc = False
            for idx, cand in enumerate(candidates):
                if cand["batch"].batch_id == b.batch_id:
                    row[idx] = 1.0
                    has_arc = True
            if has_arc:
                rows.append(row)
                rhs.append(b.quantity_kg)

        # 2. Destination demand & reception capacity constraints
        for d in destinations:
            row = np.zeros(n_vars, dtype=float)
            has_arc = False
            for idx, cand in enumerate(candidates):
                if cand["dest"].dest_id == d.dest_id:
                    row[idx] = 1.0
                    has_arc = True
            if has_arc:
                cap_limit = (
                    min(d.demand_kg, d.max_reception_capacity_kg)
                    if d.max_reception_capacity_kg is not None
                    else d.demand_kg
                )
                rows.append(row)
                rhs.append(cap_limit)

        # 3. Route vehicle capacity constraints
        for r in routes:
            row = np.zeros(n_vars, dtype=float)
            has_arc = False
            for idx, cand in enumerate(candidates):
                if (cand["route"].origin, cand["route"].dest_id) == (r.origin, r.dest_id):
                    row[idx] = 1.0
                    has_arc = True
            if has_arc:
                rows.append(row)
                rhs.append(r.vehicle_capacity_kg)

        a_ub = np.vstack(rows) if rows else None
        b_ub = np.array(rhs, dtype=float) if rhs else None

        res = linprog(c, A_ub=a_ub, b_ub=b_ub, bounds=(0, None), method="highs")
        if not res.success or res.x is None:
            return []

        return [
            (cand, round(float(qty), 2))
            for cand, qty in zip(candidates, res.x)
            if qty > 1e-4
        ]

    def solve(
        self,
        batches: List[Batch],
        destinations: List[Destination],
        routes: List[TransportRoute],
    ) -> OptimizationSummary:
        start_ns = time.perf_counter_ns()

        route_map = {(r.origin, r.dest_id): r for r in routes}
        remaining_demand = {
            d.dest_id: (
                min(d.demand_kg, d.max_reception_capacity_kg)
                if d.max_reception_capacity_kg is not None
                else d.demand_kg
            )
            for d in destinations
        }
        remaining_route_cap = {(r.origin, r.dest_id): r.vehicle_capacity_kg for r in routes}
        total_inventory_kg = sum(b.quantity_kg for b in batches)

        candidates = self._build_candidates(batches, destinations, route_map)
        batch_remaining_qty = {b.batch_id: b.quantity_kg for b in batches}
        allocations: List[AllocationDecision] = []
        dispatched_kg = 0.0
        spoiled_kg = 0.0
        fulfilled_kg = 0.0

        if self.solver_mode == "lp_highs":
            planned_dispatch = self._allocate_lp(candidates, batches, destinations, routes)
        else:
            planned_dispatch = []
            for cand in candidates:
                b: Batch = cand["batch"]
                d: Destination = cand["dest"]
                r: TransportRoute = cand["route"]

                avail_b = batch_remaining_qty[b.batch_id]
                avail_d = remaining_demand[d.dest_id]
                avail_cap = remaining_route_cap[(r.origin, r.dest_id)]

                alloc_qty = min(avail_b, avail_d, avail_cap)
                if alloc_qty <= 0:
                    continue

                batch_remaining_qty[b.batch_id] -= alloc_qty
                remaining_demand[d.dest_id] -= alloc_qty
                remaining_route_cap[(r.origin, r.dest_id)] -= alloc_qty
                planned_dispatch.append((cand, alloc_qty))

        if self.solver_mode == "lp_highs":
            for cand, alloc_qty in planned_dispatch:
                batch_remaining_qty[cand["batch"].batch_id] -= alloc_qty

        for cand, alloc_qty in planned_dispatch:
            b = cand["batch"]
            d = cand["dest"]
            r = cand["route"]

            clearance = cand["clearance_rate"]
            sold_qty = alloc_qty * clearance
            unsold_spoiled_qty = alloc_qty * (1.0 - clearance)

            spoiled_kg += unsold_spoiled_qty
            fulfilled_kg += sold_qty
            dispatched_kg += alloc_qty

            gross = sold_qty * cand["effective_price"]
            t_cost = alloc_qty * r.cost_per_kg
            p_cost = alloc_qty * b.base_cost_per_kg
            net = gross - t_cost - p_cost

            reason = (
                f"FreshRoute: Margin \u20b9{cand['unit_margin']:.1f}/kg. "
                f"Buffer {cand['buffer_hrs']:.1f}h. {cand['discount_reason']}."
            )

            allocations.append(
                AllocationDecision(
                    batch_id=b.batch_id,
                    dest_id=d.dest_id,
                    commodity=b.commodity,
                    quantity_kg=alloc_qty,
                    sold_qty_kg=round(sold_qty, 2),
                    clearance_rate=clearance,
                    travel_time_hrs=r.travel_time_hrs,
                    remaining_shelf_life_hrs=b.remaining_shelf_life_hrs,
                    shelf_life_buffer_hrs=round(cand["buffer_hrs"], 1),
                    base_price_per_kg=d.selling_price_per_kg,
                    discount_pct=cand["discount_pct"],
                    final_price_per_kg=round(cand["effective_price"], 2),
                    transport_cost_per_kg=r.cost_per_kg,
                    base_cost_per_kg=b.base_cost_per_kg,
                    gross_revenue=round(gross, 2),
                    transport_cost_total=round(t_cost, 2),
                    procurement_cost_total=round(p_cost, 2),
                    net_profit=round(net, 2),
                    reason=reason,
                )
            )

        # Unallocated inventory expiring in warehouse within 24h planning horizon
        for b in batches:
            rem = max(0.0, batch_remaining_qty[b.batch_id])
            if rem > 0 and b.remaining_shelf_life_hrs <= 24.0:
                spoiled_kg += rem

        unsold_unspoiled = max(0.0, total_inventory_kg - dispatched_kg - spoiled_kg)

        tot_gross = sum(a.gross_revenue for a in allocations)
        tot_trans = sum(a.transport_cost_total for a in allocations)
        tot_proc = sum(a.procurement_cost_total for a in allocations)
        waste_penalty = spoiled_kg * self.spoilage_penalty_per_kg
        net_profit = tot_gross - tot_trans - tot_proc - waste_penalty

        total_demand = sum(d.demand_kg for d in destinations)
        fulfillment_pct = (fulfilled_kg / total_demand * 100.0) if total_demand > 0 else 0.0
        spoilage_pct = (spoiled_kg / total_inventory_kg * 100.0) if total_inventory_kg > 0 else 0.0
        elapsed_ms = (time.perf_counter_ns() - start_ns) / 1_000_000.0

        return OptimizationSummary(
            strategy_name="FreshRoute Intelligent Engine",
            total_inventory_kg=total_inventory_kg,
            dispatched_kg=round(dispatched_kg, 1),
            spoiled_kg=round(spoiled_kg, 1),
            unsold_unspoiled_kg=round(unsold_unspoiled, 1),
            spoilage_rate_pct=round(spoilage_pct, 2),
            demand_fulfillment_pct=round(fulfillment_pct, 2),
            total_gross_revenue=round(tot_gross, 2),
            total_transport_cost=round(tot_trans, 2),
            total_procurement_cost=round(tot_proc, 2),
            disposal_waste_cost=round(waste_penalty, 2),
            net_profit=round(net_profit, 2),
            solve_time_ms=round(elapsed_ms, 3),
            allocations=allocations,
        )

    def evaluate_uncertainty(
        self,
        batches: List[Batch],
        destinations: List[Destination],
        routes: List[TransportRoute],
        runs: int = 50,
        shelf_life_noise_pct: float = 0.10,
        transit_noise_pct: float = 0.15,
        demand_noise_pct: float = 0.10,
        seed: int = 42,
    ) -> UncertaintyReport:
        """
        PS 07 Optional Extension: Monte Carlo sensitivity evaluation across uncertainty ranges
        in remaining shelf life, highway travel times, and destination demand.
        """
        rng = np.random.default_rng(seed)
        profits: List[float] = []
        spoilages: List[float] = []
        fulfillments: List[float] = []

        for _ in range(runs):
            sim_batches = [
                b.model_copy(
                    update={
                        "remaining_shelf_life_hrs": max(
                            1.0,
                            round(
                                b.remaining_shelf_life_hrs
                                * float(rng.uniform(1.0 - shelf_life_noise_pct, 1.0 + shelf_life_noise_pct)),
                                2,
                            ),
                        )
                    }
                )
                for b in batches
            ]
            sim_routes = [
                r.model_copy(
                    update={
                        "travel_time_hrs": max(
                            0.5,
                            round(
                                r.travel_time_hrs
                                * float(rng.uniform(1.0 - transit_noise_pct, 1.0 + transit_noise_pct)),
                                2,
                            ),
                        )
                    }
                )
                for r in routes
            ]
            sim_dests = [
                d.model_copy(
                    update={
                        "demand_kg": max(
                            0.0,
                            round(
                                d.demand_kg
                                * float(rng.uniform(1.0 - demand_noise_pct, 1.0 + demand_noise_pct)),
                                2,
                            ),
                        )
                    }
                )
                for d in destinations
            ]

            res = self.solve(sim_batches, sim_dests, sim_routes)
            profits.append(res.net_profit)
            spoilages.append(res.spoilage_rate_pct)
            fulfillments.append(res.demand_fulfillment_pct)

        arr_p = np.array(profits, dtype=float)
        return UncertaintyReport(
            runs=runs,
            mean_net_profit=round(float(np.mean(arr_p)), 2),
            std_net_profit=round(float(np.std(arr_p)), 2),
            p5_net_profit=round(float(np.percentile(arr_p, 5)), 2),
            p95_net_profit=round(float(np.percentile(arr_p, 95)), 2),
            mean_spoilage_pct=round(float(np.mean(spoilages)), 2),
            mean_fulfillment_pct=round(float(np.mean(fulfillments)), 2),
        )

"""engine/models.py: Core data schemas with strict constraint and security validations."""

from typing import List, Optional
from pydantic import BaseModel, ConfigDict, Field, model_validator


class Batch(BaseModel):
    """Represents a perishable inventory batch at an origin hub."""

    model_config = ConfigDict(str_strip_whitespace=True, validate_assignment=True)

    batch_id: str = Field(min_length=1, max_length=64, description="Unique batch identifier")
    commodity: str = Field(min_length=1, max_length=64, description="Produce commodity type")
    quantity_kg: float = Field(gt=0.0, le=1_000_000.0, description="Quantity in kilograms")
    remaining_shelf_life_hrs: float = Field(
        gt=0.0, le=8760.0, description="Estimated remaining shelf life in hours"
    )
    base_cost_per_kg: float = Field(
        ge=0.0, le=100_000.0, description="Cost of procurement/production per kg"
    )
    origin: str = Field(default="Warehouse-Central", min_length=1, max_length=64)
    storage_temp_c: float = Field(
        default=4.0, ge=-30.0, le=60.0, description="Observed cold-chain temperature in Celsius"
    )
    harvest_date: Optional[str] = Field(default=None, max_length=32)


class Destination(BaseModel):
    """Represents a destination market (mandi/retail hub) and its demand parameters."""

    model_config = ConfigDict(str_strip_whitespace=True, validate_assignment=True)

    dest_id: str = Field(min_length=1, max_length=64, description="Unique destination ID")
    name: str = Field(min_length=1, max_length=128, description="Destination market name")
    commodity: str = Field(min_length=1, max_length=64, description="Demanded commodity")
    demand_kg: float = Field(ge=0.0, le=1_000_000.0, description="Maximum demand at destination")
    selling_price_per_kg: float = Field(
        gt=0.0, le=100_000.0, description="Contracted or market selling price per kg"
    )
    handling_time_hrs: float = Field(
        default=2.0, ge=0.0, le=168.0, description="Unloading and dock-to-shelf handling time"
    )
    max_reception_capacity_kg: Optional[float] = Field(
        default=None, gt=0.0, le=1_000_000.0, description="Optional dock reception capacity limit"
    )


class TransportRoute(BaseModel):
    """Represents a logistics corridor between an origin hub and destination market."""

    model_config = ConfigDict(str_strip_whitespace=True, validate_assignment=True)

    dest_id: str = Field(min_length=1, max_length=64)
    travel_time_hrs: float = Field(gt=0.0, le=720.0, description="Transit time in hours")

    vehicle_capacity_kg: float = Field(
        default=10000.0, gt=0.0, le=1_000_000.0, description="Maximum vehicle payload capacity"

class AllocationDecision(BaseModel):
    """Auditable record of a single batch-to-destination dispatch allocation."""

    batch_id: str
    dest_id: str
    commodity: str
    quantity_kg: float = Field(ge=0.0)
    sold_qty_kg: float = Field(default=0.0, ge=0.0)
    clearance_rate: float = Field(default=1.0, ge=0.0, le=1.0)
    travel_time_hrs: float = Field(ge=0.0)
    remaining_shelf_life_hrs: float = Field(ge=0.0)
    shelf_life_buffer_hrs: float
    base_price_per_kg: float = Field(ge=0.0)
    discount_pct: float = Field(ge=0.0, le=1.0)
    final_price_per_kg: float = Field(ge=0.0)
    transport_cost_per_kg: float = Field(ge=0.0)
    base_cost_per_kg: float = Field(ge=0.0)
    gross_revenue: float = Field(ge=0.0)
    transport_cost_total: float = Field(ge=0.0)
    procurement_cost_total: float = Field(ge=0.0)
    net_profit: float
    reason: str

    @model_validator(mode="after")
    def set_default_sold_qty(self) -> "AllocationDecision":
        if self.sold_qty_kg == 0.0 and self.gross_revenue > 0.0:
            self.sold_qty_kg = round(self.quantity_kg * self.clearance_rate, 2)
        return self

"""tests/test_freshroute.py: Comprehensive test suite covering constraints, LP, security, and edge cases."""

import pytest
from pydantic import ValidationError

from engine.baseline_solver import BaselineSolver
from engine.data_generator import (
    apply_disruption,
    apply_temperature_excursion,
    get_standard_scenario,
)

def test_freshroute_outperforms_baseline():
    batches, dests, routes = get_standard_scenario()
    baseline = BaselineSolver().solve(batches, dests, routes)
    fresh = FreshRouteSolver().solve(batches, dests, routes)

    assert fresh.net_profit > baseline.net_profit, "FreshRoute should generate higher net profit"
    assert fresh.spoilage_rate_pct <= baseline.spoilage_rate_pct, "FreshRoute should reduce or match spoilage"
    assert fresh.net_profit == pytest.approx(502497.0, rel=1e-4)
    assert baseline.net_profit == pytest.approx(321690.0, rel=1e-4)


def test_zero_in_transit_spoilage_for_freshroute():
    batches, dests, routes = get_standard_scenario()
    solver = FreshRouteSolver()
    fresh = solver.solve(batches, dests, routes)

    for alloc in fresh.allocations:
        assert alloc.shelf_life_buffer_hrs >= solver.min_safe_buffer_hrs - 1e-5, (
            f"Batch {alloc.batch_id} violated minimum safe buffer upon arrival!"
        )


def test_capacity_and_demand_constraints_respected():
    batches, dests, routes = get_standard_scenario()
    fresh = FreshRouteSolver().solve(batches, dests, routes)

    dest_allocs = {}
    route_allocs = {}
    for a in fresh.allocations:
        dest_allocs[a.dest_id] = dest_allocs.get(a.dest_id, 0.0) + a.quantity_kg
        route_allocs[a.dest_id] = route_allocs.get(a.dest_id, 0.0) + a.quantity_kg

    for d in dests:
        allocated = dest_allocs.get(d.dest_id, 0.0)
        assert allocated <= d.demand_kg + 1e-4, f"Demand constraint violated for destination {d.dest_id}"

    for r in routes:
        allocated = route_allocs.get(r.dest_id, 0.0)
        assert allocated <= r.vehicle_capacity_kg + 1e-4, f"Vehicle capacity violated for route {r.dest_id}"


def test_disruption_handling_robustness():
    """Verify solver stays robust when a 6-hour delay occurs across all routes."""
    batches, dests, routes = get_standard_scenario()
    bad_routes, bad_dests = apply_disruption(routes, dests, delay_hours=6.0, demand_shock_pct=-0.25)

    fresh = FreshRouteSolver().solve(batches, bad_dests, bad_routes)
    assert fresh.net_profit > 0, "Engine should stay profitable or preserve capital even under severe delays"


def test_lp_highs_solver_mode():
    """Verify SciPy HiGHS Linear Programming mode satisfies all constraints and beats baseline."""
    batches, dests, routes = get_standard_scenario()
    baseline = BaselineSolver().solve(batches, dests, routes)
    lp_res = FreshRouteSolver(solver_mode="lp_highs").solve(batches, dests, routes)

    assert lp_res.net_profit >= baseline.net_profit
    assert lp_res.spoilage_rate_pct == 0.0
    assert lp_res.demand_fulfillment_pct > baseline.demand_fulfillment_pct

def test_clearance_elasticity_and_fulfillment_alignment():
    """Verify fulfillment_pct accurately reflects sold_qty_kg when candidates are skipped."""
    batches, dests, routes = get_standard_scenario()
    solver = FreshRouteSolver(max_discount_pct=0.05)
    res = solver.solve(batches, dests, routes)

    total_demand = sum(d.demand_kg for d in dests)
    expected_fulfillment = round(sum(a.sold_qty_kg for a in res.allocations) / total_demand * 100.0, 2)
    assert res.demand_fulfillment_pct == pytest.approx(expected_fulfillment, abs=1e-2)
